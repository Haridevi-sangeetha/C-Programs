#include<stdio.h>

int main(){
    int num;
    scanf("%d",&num);
    num = num+1;
    int count=0;
    while(1){
        int original=num;
        int rev=0;
        while(original!=0){
            int id = original%10;
            rev=(rev*10)+id;
            original/=10;
        }
        if(rev==num){
            printf("%d ",num);
            count++;
            if(count==5){
                break;
            }
        }
        num++;
    }
}
