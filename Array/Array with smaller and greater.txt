#include<stdio.h>

int main(){
    int size;
    scanf("%d",&size);
    int arr[size];
    for(int i=0;i<size;i++){
        scanf("%d",&arr[i]);
    }
    for(int i=0;i<size;i++){
        for(int j=i+1;j<size;j++){
            if(arr[i]>arr[j]){
                int temp=arr[i];
                arr[i]=arr[j];
                arr[j]=temp;
            }
        }
    }
    int left=0;
    int right=size-1;
    int count=0;
    for(int i=0;i<size;i++){
        if(left == right){
            printf("%d",arr[left]);
        }
        else if(left<=right){
            printf("%d ",arr[left]);
            printf("%d ",arr[right]);
        }
        left++;
        right--;
    }
    return 0;
}
