#include <stdio.h>

int main() {
    char str[] = "programming";
    int count[256] = {0};

    for(int i = 0; str[i] != '\0'; i++)
        count[str[i]]++;

    for(int i = 0; str[i] != '\0'; i++) {
        if(count[str[i]] == 1) {
            printf("First Non-Repeating: %c", str[i]);
            break;
        }
    }

    return 0;
}
