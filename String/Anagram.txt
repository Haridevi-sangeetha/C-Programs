#include <stdio.h>

int main() {
    char str1[] = "listen";
    char str2[] = "silent";
    int count1[26] = {0}, count2[26] = {0};
    int i = 0;

    while(str1[i] != '\0') {
        count1[str1[i] - 'a']++;
        count2[str2[i] - 'a']++;
        i++;
    }

    for(i = 0; i < 26; i++) {
        if(count1[i] != count2[i]) {
            printf("Not Anagram");
            return 0;
        }
    }

    printf("Anagram");
    return 0;
}
