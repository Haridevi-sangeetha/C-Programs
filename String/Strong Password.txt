#include <stdio.h>
#include <string.h>

int main(){
    char s[100];
    scanf("%s",s);
    int len = strlen(s);
    int l_count=0;
    int u_count =0;
    int n_count=0;
    int spc = 0;
    int valid = 1;
    for(int i=0;i<len;i++){
        // printf("%c ",s[i]);
        
        if(s[i]>='a' && s[i]<='z'){
            l_count++;
        }
        else if(s[i]>='A' && s[i]<='Z'){
            u_count++;
        }
        else if(s[i]>='0' && s[i]<='9'){
            n_count++;
        }
        else if(s[i]=='@' ||s[i]=='#'&& valid || s[i] =='$' || s[i] =='%' ||s[i] == '&'){
            spc++;
        }
    }
    if(l_count>=1 && n_count>=1 && spc>=1 && u_count++ &&len==8){
        printf("Strong password");
    }
    else{
        printf("week password");
    }
}
