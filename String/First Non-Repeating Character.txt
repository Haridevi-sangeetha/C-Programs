#include<stdio.h>

int main(){
    char str[100];
    scanf("%s",str); 
    int digit=0;
    int i=0;
    while(str[i]!='\0'){
        if(str[i]!='\n'){
            digit++;
        }
        i++;
    }
    // printf("%d",digit);
    int count;
    for(int i=0;i<digit;i++){
        count=0;
        for(int j=0;j<digit;j++){
            if(str[i]==str[j]){
                count+=1;
            }
        }
        printf("str[i] : %c count : %d\n",str[i],count);
        if(count==1){
            printf("%c",str[i]);
            return 0;
        }
    }
    return 0;
}
